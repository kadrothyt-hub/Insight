class AppSettings extends ChangeNotifier {
  Locale locale = const Locale('fa');
  ThemeMode themeMode = ThemeMode.dark;
  final Set<String> saved = {};

  void setLocale(Locale l) { locale = l; notifyListeners(); }
  void setTheme(ThemeMode m) { themeMode = m; notifyListeners(); }
  void toggleSaved(String id) {
    saved.contains(id) ? saved.remove(id) : saved.add(id);
    notifyListeners();
  }
}

class SettingsScope extends InheritedNotifier<AppSettings> {
  const SettingsScope({super.key, required AppSettings settings, required super.child})
      : super(notifier: settings);
  static AppSettings of(BuildContext c) =>
      c.dependOnInheritedWidgetOfExactType<SettingsScope>()!.notifier!;
}


const neon = Color(0xFF2F7BFF);

Color cardColor(BuildContext c) =>
    Theme.of(c).brightness == Brightness.dark ? const Color(0xFF121B2E) : Colors.white;

Color mutedText(BuildContext c) =>
    Theme.of(c).brightness == Brightness.dark ? const Color(0xFF8A97B3) : const Color(0xFF64708A);

ThemeData buildTheme(Brightness b, bool fa) {
  final dark = b == Brightness.dark;
  final bg = dark ? const Color(0xFF070C18) : const Color(0xFFF3F6FC);
  final base = ThemeData(
    brightness: b,
    useMaterial3: true,
    colorScheme: ColorScheme.fromSeed(seedColor: neon, brightness: b),
  );
  final text = fa
      ? GoogleFonts.vazirmatnTextTheme(base.textTheme)
      : GoogleFonts.interTextTheme(base.textTheme);
  return base.copyWith(
    textTheme: text,
    scaffoldBackgroundColor: bg,
    appBarTheme: AppBarTheme(
      backgroundColor: bg,
      elevation: 0,
      scrolledUnderElevation: 0,
      centerTitle: false,
    ),
    navigationBarTheme: NavigationBarThemeData(
      backgroundColor: dark ? const Color(0xFF0C1424) : Colors.white,
      indicatorColor: neon.withAlpha(40),
      labelTextStyle: WidgetStatePropertyAll(
        TextStyle(fontSize: 11, fontFamily: text.bodyMedium?.fontFamily),
      ),
    ),
  );
}

BoxDecoration cardDeco(BuildContext c) => BoxDecoration(
      color: cardColor(c),
      borderRadius: BorderRadius.circular(20),
      boxShadow: [
        BoxShadow(
          color: Colors.black.withAlpha(Theme.of(c).brightness == Brightness.dark ? 60 : 18),
          blurRadius: 18,
          offset: const Offset(0, 6),
        ),
      ],
    );

class CatChip extends StatelessWidget {
  final String cat;
  const CatChip(this.cat, {super.key});
  @override
  Widget build(BuildContext context) {
    final c = categoryColors[cat] ?? neon;
    return Container(
      padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 4),

    );
  }
}

class _ChatPageState extends State<ChatPage> {
  final ctrl = TextEditingController();
  final msgs = <_Msg>[];
  String? tool;
  static const tools = ['DEEP RESEARCH', 'FACT CHECK', 'HIDDEN CONNECTIONS', 'FUTURE ANALYSIS', 'INSIGHT GENERATION'];

  void send() {
    final text = ctrl.text.trim();
    if (text.isEmpty) return;
    setState(() { msgs.add(_Msg(tool == null ? text : '[$tool] $text', true)); ctrl.clear(); });
    // TODO phase 2: call your backend (which calls the AI API). Never ship an API key inside the app.
    Future.delayed(const Duration(milliseconds: 600), () {
      if (!mounted) return;
      setState(() => msgs.add(_Msg(t(context, 'chatstub'), false)));
    });
  }

  @override
  Widget build(BuildContext context) {
    return Column(children: [
      Padding(
        padding: const EdgeInsets.fromLTRB(16, 12, 16, 4),
        child: Align(
          alignment: AlignmentDirectional.centerStart,
          child: Text('INSIGHT CHAT', style: TextStyle(color: neon, fontWeight: FontWeight.w900, letterSpacing: 2)),
        ),
      ),
      SizedBox(
        height: 46,
        child: ListView(
          scrollDirection: Axis.horizontal,
          padding: const EdgeInsets.symmetric(horizontal: 12),
          children: [
            for (final x in tools)
              Padding(
                padding: const EdgeInsets.symmetric(horizontal: 4),
                child: FilterChip(
                  label: Text(x, style: const TextStyle(fontSize: 11)),
                  selected: tool == x,
                  onSelected: (v) => setState(() => tool = v ? x : null),
                ),
              ),
          ],
        ),
      ),
      Expanded(
        child: ListView(padding: const EdgeInsets.all(16), children: [
          _Bubble(t(context, 'chatintro'), false),
          for (final m in msgs) _Bubble(m.text, m.user),
        ]),
      ),
      Padding(
        padding: const EdgeInsets.fromLTRB(12, 4, 12, 12),
        child: Row(children: [
          Expanded(
            child: TextField(
              controller: ctrl,
              onSubmitted: (_) => send(),
              decoration: InputDecoration(
                hintText: t(context, 'ask'),
                filled: true,
                fillColor: cardColor(context),
                border: OutlineInputBorder(borderRadius: BorderRadius.circular(24), borderSide: BorderSide.none),
              ),
            ),
          ),
          const SizedBox(width: 8),
          IconButton.filled(onPressed: send, icon: const Icon(Icons.arrow_upward_rounded)),
        ]),
      ),
    ]);
  }

                  const SizedBox(width: 2),
                  const Text('OFFICIAL', style: TextStyle(color: neon, fontSize: 10, fontWeight: FontWeight.w800)),
                ],
              ]),
              Text('@${p.handle} · ${p.time.of(context)}', style: TextStyle(color: mutedText(context), fontSize: 12)),

        ]),
        const SizedBox(height: 12),
        Text(p.text.of(context), style: const TextStyle(height: 1.8)),
        const SizedBox(height: 10),
        Text(p.tag, style: TextStyle(color: mutedText(context), fontSize: 11, fontWeight: FontWeight.w700, letterSpacing: 1)),

          IconButton(
            icon: Icon(liked ? Icons.favorite_rounded : Icons.favorite_border_rounded, color: liked ? Colors.redAccent : null),
            onPressed: () => setState(() => liked = !liked),
          ),
          Text('${p.likes + (liked ? 1 : 0)}'),
          const SizedBox(width: 8),
          IconButton(icon: const Icon(Icons.chat_bubble_outline_rounded), onPressed: () {}),
          IconButton(icon: const Icon(Icons.repeat_rounded), onPressed: () {}),
          Text('${p.reposts}'),
          const Spacer(),
          IconButton(
            icon: Icon(saved ? Icons.bookmark_rounded : Icons.bookmark_border_rounded),
            onPressed: () => setState(() => saved = !saved),
          ),
        ]),
      ]),
    );
  }
}

class InsightApp extends StatefulWidget {
  const InsightApp({super.key});
  @override
  State<InsightApp> createState() => _InsightAppState();
}

class _InsightAppState extends State<InsightApp> {
  final settings = AppSettings();

  @override
  Widget build(BuildContext context) {
    return ListenableBuilder(
      listenable: settings,
      builder: (_, __) {
        final fa = settings.locale.languageCode == 'fa';
        return SettingsScope(
          settings: settings,
          child: MaterialApp(
            debugShowCheckedModeBanner: false,
            title: 'INSIGHT',
            locale: settings.locale,
            supportedLocales: const [Locale('fa'), Locale('en')],
            localizationsDelegates: const [
              GlobalMaterialLocalizations.delegate,
              GlobalWidgetsLocalizations.delegate,
              GlobalCupertinoLocalizations.delegate,
            ],
            themeMode: settings.themeMode,
            theme: buildTheme(Brightness.light, fa),
            darkTheme: buildTheme(Brightness.dark, fa),
            home: const Shell(),
          ),
        );
      },
    );
  }
}
